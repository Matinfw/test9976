# تابع برای استخراج متن از فایل PDF
def extract_text_from_pdf(pdf_file):
    """
    Extracts text from a PDF file-like object.
    """
    try:
        pdf_reader = PdfReader(pdf_file)
        text = ""
        for page in pdf_reader.pages:
            page_text = page.extract_text()
            if page_text:
                text += page_text + "\n"
        return text
    except Exception as e:
        return f"خطا در استخراج متن از PDF: {str(e)}"

# تابع برای خلاصه‌سازی متن با استفاده از Gemini API
def summarize_text(text, api_key, max_length=500):
    """
    Summarizes the given text using the Google Gemini API.
    Uses the provided api_key for configuration.
    """
    try:
        # Configure the generative AI library with the API key from the UI
        genai.configure(api_key=api_key)

        # Use a model name that is available and supports generateContent
        # "gemini-1.5-flash-latest" is a common and fast model.
        model_name = "gemini-1.5-flash-latest"

        model = genai.GenerativeModel(model_name)

        prompt = f"""
        شما یک متخصص خلاصه‌سازی اسناد حرفه‌ای هستید. متن زیر را به زبان فارسی خلاصه کنید.
        خلاصه باید حداکثر {max_length} کلمه باشد، منسجم و شامل نکات کلیدی متن باشد.
        متن:
        {text}
        """
        response = model.generate_content(prompt)
        return response.text
    except Exception as e:
        return f"❌ خطا در خلاصه‌سازی متن: {str(e)}"

# تابع برای تبدیل متن به صوت و ذخیره آن
def text_to_audio(text, output_file="summary_audio.mp3"):
    """
    Converts text to speech using gTTS and saves it to an MP3 file.
    Uses Arabic ('ar') as a fallback language for Persian if 'fa' is not supported.
    """
    try:
        # *** FIX: Use Arabic ('ar') as a fallback for Persian ('fa') ***
        # gTTS might not fully support 'fa'. 'ar' is often a supported alternative.
        tts = gtts.gTTS(text, lang="ar") # Change lang to 'ar'
        tts.save(output_file)

        return output_file
    except Exception as e:
        # Return a more specific error message if it's a language issue
        if "Language not supported" in str(e) or "fa" in str(e).lower():
             return f"❌ خطا در تبدیل متن به صوت: زبان 'fa' توسط gTTS پشتیبانی نمی‌شود. لطفاً از یک زبان دیگر (مثل 'ar') استفاده کنید. پیام خطا: {str(e)}"
        else:
            return f"❌ خطا در تبدیل متن به صوت: {str(e)}"

# ایجاد رابط کاربری تعاملی با استفاده از ipywidgets
def create_ui():
    """
    Creates and displays the interactive user interface using ipywidgets.
    """
    # Input box for the API key (still needed for genai.configure)
    api_key_input = widgets.Text(
        value="",
        placeholder="کلید API Google Gemini خود را وارد کنید",
        description="کلید API:",
        layout={'width': '500px'},
        style={'description_width': 'initial'}
    )

    # File upload button, accepting only PDF files
    upload_button = widgets.FileUpload(
        accept=".pdf",
        multiple=False,
        description="آپلود فایل PDF"
    )

    # Button to start the process
    process_button = widgets.Button(
        description="شروع خلاصه‌سازی و تبدیل به صوت",
        button_style="success",
        tooltip="برای شروع فرآیند کلیک کنید",
        icon="play"
    )

    # Output widget to display status and results
    output = widgets.Output()

    # Function to handle the button click event
    def on_process_button_clicked(b):
        with output:
            output.clear_output()
            print("در حال شروع فرآیند...")

            # --- Validation ---
            if not api_key_input.value:
                print("⚠️ لطفاً کلید API را وارد کنید.")
                return
            if not upload_button.value:
                print("⚠️ لطفاً یک فایل PDF آپلود کنید.")
                return

            # --- File Handling ---
            try:
                uploaded_file_info = list(upload_button.value.values())[0]
                file_content = uploaded_file_info['content']
                pdf_file_io = io.BytesIO(file_content)
                print(f"✅ فایل '{uploaded_file_info['metadata']['name']}' با موفقیت آپلود شد.")
            except Exception as e:
                print(f"❌ خطا در دسترسی به فایل آپلود شده: {e}")
                return

            # --- Text Extraction ---
            print("⏳ در حال استخراج متن از PDF...")
            text = extract_text_from_pdf(pdf_file_io)
            if "خطا" in text:
                print(f"❌ {text}")
                return
            print("✅ متن با موفقیت استخراج شد.")


            # --- Summarization ---
            print("⏳ در حال خلاصه‌سازی متن با استفاده از Gemini...")
            # Use the API key from the UI input
            summary = summarize_text(text, api_key_input.value)
            if "خطا" in summary:
                print(f"❌ {summary}")
                print("💡 نکات: مطمئن شوید کلید API صحیح است و به مدل دسترسی دارید.")
                return
            print("✅ خلاصه‌سازی با موفقیت انجام شد.")

            # --- Display Summary ---
            print("\n--- خلاصه متن ---")
            display(widgets.Textarea(
                value=summary,
                description='خلاصه:',
                disabled=True,
                layout={'width': 'auto', 'height': '200px'}
            ))
            print("-----------------")


            # --- Text-to-Audio Conversion ---
            print("\n⏳ در حال تبدیل خلاصه به فایل صوتی...")
            audio_file_path = text_to_audio(summary)
            if "خطا" in audio_file_path:
                print(f"❌ {audio_file_path}")
                print("💡 نکات: ممکن است خلاصه‌سازی خالی یا خیلی کوتاه بوده باشد یا مشکل gTTS در تبدیل صوت باشد.")
                return
            print(f"✅ فایل صوتی با موفقیت ایجاد شد: '{audio_file_path}'")

            # --- Display Audio Player ---
            print("\n--- پخش فایل صوتی ---")
            try:
                display(Audio(audio_file_path, embed=True))
            except Exception as e:
                 print(f"❌ خطا در نمایش پخش‌کننده صوتی: {e}")


            # --- Provide Download Link ---
            print("\n--- دانلود فایل صوتی ---")
            print(f"برای دانلود فایل '{audio_file_path}' روی لینک زیر کلیک کنید:")
            try:
                 files.download(audio_file_path)
            except Exception as e:
                 print(f"❌ خطا در ارائه لینک دانلود: {e}")


            print("\n✨ فرآیند به پایان رسید.")


    process_button.on_click(on_process_button_clicked)

    print("💡 لطفاً کلید API Google Gemini خود را وارد کرده و فایل PDF را آپلود کنید.")
    display(api_key_input)
    display(upload_button)
    display(process_button)
    display(output)

# --- Run the UI ---
create_ui()
